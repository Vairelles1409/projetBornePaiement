-- ========================================================
-- GESTION DES COMMANDES ET PAIEMENTS
-- ========================================================

CREATE TABLE `coupons` (
  `id_coupon` int(11) NOT NULL AUTO_INCREMENT,
  `valeur` int(11) NOT NULL,
  `validity_date` date DEFAULT NULL,
  `status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id_coupon`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `commandes` (
  `id_order` int(11) NOT NULL AUTO_INCREMENT,
  `id_client` int(11) DEFAULT NULL,
  `total_price` double NOT NULL,
  `order_date` datetime(6) DEFAULT NULL,
  PRIMARY KEY (`id_order`),
  CONSTRAINT `FK_Commande_Client` FOREIGN KEY (`id_client`) REFERENCES `clients` (`id_personne`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `lignes_commande` (
  `id_ligne` int(11) NOT NULL AUTO_INCREMENT,
  `id_order` int(11) DEFAULT NULL,
  `id_plat` int(11) DEFAULT NULL,
  `price_ligne` double NOT NULL,
  `qty_commande` int(11) NOT NULL,
  PRIMARY KEY (`id_ligne`),
  CONSTRAINT `FK_Ligne_Commande` FOREIGN KEY (`id_order`) REFERENCES `commandes` (`id_order`),
  CONSTRAINT `FK_Ligne_Plat` FOREIGN KEY (`id_plat`) REFERENCES `plats` (`id_plat`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `paiements` (
  `id_payment` int(11) NOT NULL AUTO_INCREMENT,
  `id_order` int(11) DEFAULT NULL, -- Peut être NULL au début, ou Unique
  `id_client` int(11) DEFAULT NULL,
  `final_amount` double NOT NULL,
  `payment_date` datetime(6) DEFAULT NULL,
  `card_network` varchar(255) DEFAULT NULL, -- ex: VISA, MASTERCARD
  `status` varchar(255) DEFAULT NULL,       -- ex: SUCCESS, FAILED
  PRIMARY KEY (`id_payment`),
  UNIQUE KEY `UK_Payment_Order` (`id_order`),
  CONSTRAINT `FK_Paiement_Client` FOREIGN KEY (`id_client`) REFERENCES `clients` (`id_personne`),
  CONSTRAINT `FK_Paiement_Commande` FOREIGN KEY (`id_order`) REFERENCES `commandes` (`id_order`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `tickets` (
  `id_ticket` int(11) NOT NULL AUTO_INCREMENT,
  `id_payment` int(11) DEFAULT NULL,
  `content` tinytext DEFAULT NULL,
  PRIMARY KEY (`id_ticket`),
  UNIQUE KEY `UK_Ticket_Payment` (`id_payment`),
  CONSTRAINT `FK_Ticket_Paiement` FOREIGN KEY (`id_payment`) REFERENCES `paiements` (`id_payment`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
